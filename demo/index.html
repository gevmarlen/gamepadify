const pad = createGamepadify({
  navigation: { confirm: 'xbox:a' },
  showHints: true,
});
pad.start();

xbox:a, ps:cross and switch:b all describe the bottom face button — south in the platform-neutral API. Prompts use your controller’s symbols.
